    def make_llm_client(cfg: dict) -> Any:
    """Build a chat client for the provider named in cfg['ask_ai_provider']."""
    apply_provider_config(cfg)
    preferred = cfg.get("ask_ai_provider") or "llama-server"
    client = LLMClient(
        prefer_lm_studio=(preferred == "LM Studio"),
        preferred_provider=preferred,
        auto_detect=True,
    )
    if preferred == "OpenAI-compatible" and cfg.get("openai_compat_model"):
        client.model = cfg["openai_compat_model"]
    elif preferred == "LM Studio" and cfg.get("lm_studio_model"):
        client.model = cfg["lm_studio_model"]
    elif preferred == "Ollama" and cfg.get("ollama_model"):
        client.model = cfg["ollama_model"]
    elif preferred == "llama-server":
        name = str(cfg.get("chat_model_name") or "").strip()
        path = str(cfg.get("chat_model_path") or "").strip()
        if not name and path:
            name = os.path.basename(path)
        LLMClient._llama_model_name = name
        client.model = client.refresh_llama_model()
    return client

    def live_vs_corpus(
    embeddings_url: str = "",
    *,
    profile_model_path: str = "",
    profile_model_id: str = "",
    db_path: str = "",
) -> Dict[str, Any]:
    """Compare live :8080, optional profile GGUF, and kb_embedding_corpus.

    Hard mismatch is live server vs KB corpus (what actually embeds vs stored
    vectors). Profile path is advisory: a directory-only ``model_path`` is
    ignored unless ``model_id`` / a joined file yields a real GGUF identity.
    """
    corpus = read_corpus(db_path)
    url = embeddings_url or (corpus or {}).get("embeddings_url") or (
        "http://localhost:8080/embeddings"
    )
    live = probe_live_model(url)
    corpus_path = str((corpus or {}).get("model_path") or "")
    live_path = str(live.get("model_path") or "")
    profile_path = resolve_gguf_identity(profile_model_path, profile_model_id)
    out: Dict[str, Any] = {
        "corpus": corpus,
        "live": live,
        "profile_model_path": profile_path or str(profile_model_path or ""),
        "ok": True,
        "mismatch": False,
        "profile_mismatch": False,
        "port_occupied_wrong_model": False,
        "message": "",
    }
    if not corpus_path:
        return out
    # Blocking: live GGUF must match the wiki-scan corpus when the server is up.
    if live.get("reachable") and live_path and not models_match(live_path, corpus_path):
        out["ok"] = False
        out["mismatch"] = True
        out["port_occupied_wrong_model"] = True
    # Advisory: profile GGUF (if resolvable) differs from corpus/live.
    if profile_path and corpus_path and not models_match(profile_path, corpus_path):
        out["profile_mismatch"] = True
    if profile_path and live_path and live.get("reachable") and not models_match(
        profile_path, live_path
    ):
        out["profile_mismatch"] = True
    if out["mismatch"]:
        out["message"] = format_mismatch(
            corpus_path=corpus_path,
            live_path=live_path if live.get("reachable") else "",
            profile_path=profile_path,
            live_reachable=bool(live.get("reachable")),
        )
    elif out["profile_mismatch"]:
        out["message"] = (
            format_mismatch(
                corpus_path=corpus_path,
                live_path=live_path if live.get("reachable") else "",
                profile_path=profile_path,
                live_reachable=bool(live.get("reachable")),
            )
            + "\n\nNote: live server matches the KB; only the saved profile "
            "path/id looks stale. Wiki Scan can proceed."
        )
    return out

    def models_match(a: str, b: str) -> bool:
    """True if two GGUF identities are the same path or the same basename."""
    fa, fb = fold_path(a), fold_path(b)
    if not fa or not fb:
        return False
    if fa == fb:
        return True
    return fold_path(model_basename(a)) == fold_path(model_basename(b))

    def record_from_live_server(
    embeddings_url: str,
    *,
    embeddings_dim: Optional[int] = None,
    model_id: str = "",
    db_path: str = "",
) -> Tuple[bool, str, Optional[Dict[str, Any]]]:
    """Probe :8080 and persist kb_embedding_corpus. False if server/model unknown."""
    live = probe_live_model(embeddings_url)
    if not live.get("reachable"):
        return False, (
            "Embedding server is not answering on "
            f"{live.get('host')}:{live.get('port')}. "
            "Launch the embedding GGUF on :8080 before Wiki Scan."
        ), live
    if not live.get("model_path"):
        return False, (
            "Embedding server is up but did not report a GGUF path "
            "(/props or /v1/models). Will not scan under an unknown model."
        ), live
    row = write_corpus(
        model_path=str(live["model_path"]),
        embeddings_url=embeddings_url,
        embeddings_dim=embeddings_dim,
        model_id=model_id,
        db_path=db_path,
    )
    return True, f"Recording embeddings from {row.get('model_name') or row.get('model_path')}", row

    def resolve_gguf_identity(path: str = "", model_id: str = "") -> str:
    """Best GGUF identity from a path and/or model id.

    Embedding profiles sometimes store a *directory* in ``model_path`` and the
    filename in ``model_id``. A bare directory must not be compared as a GGUF
    (basename would be ``models`` and false-mismatch KB/live).
    """
    p = str(path or "").strip()
    mid = str(model_id or "").strip()
    if p and is_gguf_path(p):
        return p
    if p and os.path.isdir(p) and mid and is_gguf_path(mid):
        joined = os.path.join(p, model_basename(mid))
        if os.path.isfile(joined):
            return joined
        return mid
    if mid and is_gguf_path(mid):
        return mid
    if p and os.path.isfile(p):
        return p
    return ""

    def embedding_from_response(data: Any) -> List[float]:
    """Pull the first embedding out of a llama.cpp or OpenAI-style response."""
    raw = None
    if isinstance(data, list) and data:
        item = data[0]
        if isinstance(item, dict) and "embedding" in item:
            raw = item["embedding"]
        elif isinstance(item, list):
            raw = item
    elif isinstance(data, dict):
        for key in ("data", "embeddings"):
            inner = data.get(key)
            if isinstance(inner, list) and inner:
                item = inner[0]
                raw = item["embedding"] if isinstance(item, dict) else item
                break
    if raw is None:
        raise ValueError("The embedding response did not contain a vector.")
    return flatten_embedding(raw)

    def embeddings_from_response(
    data: Any, count: int
) -> Tuple[List[Optional[List[float]]], bool]:
    """One vector per input, plus whether any vector was a JSON null.

    A null vector becomes None so the rest of the batch can still be stored.
    """
    slots: List[Optional[List[float]]] = [None] * max(0, int(count))
    saw_null = False
    items: List[Any] = []
    if isinstance(data, list):
        items = data
    elif isinstance(data, dict):
        inner = data.get("data")
        if not isinstance(inner, list):
            inner = data.get("embeddings")
        if isinstance(inner, list):
            items = inner
    for pos, item in enumerate(items):
        if isinstance(item, dict):
            idx = item.get("index", pos)
            raw = item.get("embedding")
        else:
            idx = pos
            raw = item
        if not isinstance(idx, int) or isinstance(idx, bool) or idx < 0 or idx >= len(slots):
            continue
        try:
            slots[idx] = flatten_embedding(raw)
        except ValueError as exc:
            slots[idx] = None
            if "null" in str(exc).lower():
                saw_null = True
    return slots, saw_null

    def flatten_embedding(raw: Any) -> List[float]:
    """Turn a llama-server embedding payload into one finite vector.

    Nested token rows are averaged. A null component is a NaN from the server
    and is not a number that can be added into the vector.
    """
    if isinstance(raw, list) and raw and isinstance(raw[0], list):
        rows = [row for row in raw if isinstance(row, list) and row]
        if not rows:
            raise ValueError(NULL_EMBEDDING)
        dim = len(rows[0])
        acc = [0.0] * dim
        used = 0
        for row in rows:
            if len(row) != dim:
                continue
            if any(not isinstance(value, (int, float)) for value in row):
                continue
            for index, value in enumerate(row):
                acc[index] += float(value)
            used += 1
        if used == 0:
            raise ValueError(NULL_EMBEDDING)
        return [value / used for value in acc]
    if isinstance(raw, list) and raw:
        if any(not isinstance(value, (int, float)) for value in raw):
            raise ValueError(NULL_EMBEDDING)
        return [float(value) for value in raw]
    raise ValueError(NULL_EMBEDDING)

    def probe_embeddings_dim(url: str, timeout: int = 30) -> Optional[int]:
    """POST a short string to /embeddings and return vector length."""
    payload = json.dumps({"input": "spock probe"}).encode("utf-8")
    try:
        req = requests.post(
            url,
            data=payload,
            headers={"Content-Type": "application/json"},
            timeout=timeout,
        )
        if req.status_code != 200:
            return None
        return len(embedding_from_response(req.json()))
    except Exception:
        return None

    def write_embedding_env(profile: Dict[str, Any], env_path: Optional[str] = None) -> None:
    """Upsert embedding-related keys in .env (create if missing)."""
    path = env_path or str(ROOT / ".env")
    keys = {
        "EMBEDDINGS_URL": str(profile.get("embeddings_url", "")),
        "EMBEDDINGS_DIM": str(profile.get("embeddings_dim", 4096)),
        "EMBEDDINGS_TRUNCATE": str(profile.get("truncate_chars", 1500)),
        "EMBEDDINGS_BATCH_CHUNK": str(profile.get("batch_chunk", 32)),
        "EMBEDDINGS_BATCH_WORKERS": str(profile.get("batch_workers", 8)),
    }
    lines_out: List[str] = []
    if os.path.isfile(path):
        with open(path, "r", encoding="utf-8") as f:
            for line in f:
                stripped = line.strip()
                if stripped == "# Embedding profile (managed by Spock-onboarding)":
                    continue
                if not stripped or stripped.startswith("#") or "=" not in stripped:
                    lines_out.append(line.rstrip("\n"))
                    continue
                k, _, _v = stripped.partition("=")
                if k in keys:
                    continue
                lines_out.append(line.rstrip("\n"))
    while lines_out and lines_out[-1] == "":
        lines_out.pop()
    lines_out.append("")
    lines_out.append("# Embedding profile (managed by Spock-onboarding)")
    for k, v in keys.items():
        lines_out.append(f"{k}={v}")
    with open(path, "w", encoding="utf-8") as f:
        f.write("\n".join(lines_out) + "\n")

    class RagAdapterHost:
    """Own RAG API lifecycle for EmbeddingProfilePanel callbacks."""

    def __init__(
        self,
        config: Dict[str, Any],
        save_config: Callable[[], None],
        *,
        python_exe: Optional[Path] = None,
        parent: Optional[QWidget] = None,
    ):
        self.config = config
        self._save_config = save_config
        self.parent = parent
        self.panel = None
        self.python_exe = python_exe or self._default_python()

    def _default_python(self) -> Path:
        venv_py = ROOT / ".venv" / "Scripts" / "python.exe"
        if venv_py.is_file():
            return venv_py
        return Path(sys.executable)

    def _log(self, msg: str) -> None:
        if self.panel is not None and hasattr(self.panel, "activity_log"):
            try:
                self.panel.activity_log.append(msg)
            except Exception:
                pass

    def refresh_panel(self) -> None:
        if self.panel is None:
            return
        host, port = rag_bind(self.config)
        base = rag_url(host, port)
        if is_rag_up(host, port):
            self.panel.set_rag_running(True, f"{base}/v1")
        else:
            self.panel.set_rag_running(False)

    def on_profile_applied(self, profile: dict) -> None:
        from WarpCore.config_defaults import merge_embedding_profile
        from spock_rag_adapter.embedding_utils import write_embedding_env

        self.config["embedding_profile"] = merge_embedding_profile(profile)
        if self.panel is not None:
            self.config["rag_api"] = self.panel.collect_rag_api()
        write_embedding_env(self.config["embedding_profile"])
        self._save_config()
        self._log("Embedding profile saved to ship_computer.db + .env")

    def start_rag(self) -> None:
        if self.panel is None:
            return
        self.config["rag_api"] = self.panel.collect_rag_api()
        self.config.setdefault("rag_api", {})["enabled"] = True
        self._save_config()

        host, port = rag_bind(self.config)
        url = rag_url(host, port)
        script = RAG_SCRIPT
        py = self.python_exe

        if not script.is_file():
            QMessageBox.critical(self.parent, "RAG API", f"Script not found:\n{script}")
            return
        if not py.is_file():
            QMessageBox.critical(self.parent, "RAG API", f"Python not found:\n{py}")
            return

        self._log(f"Starting RAG API at {url} …")

        if is_rag_up(host, port):
            self._log(f"Already running at {url}")
            self.panel.set_rag_running(True, f"{url}/v1")
            return

        with exclusive_start("rag_api") as locked:
            existing = pids_running_script(script)
            if existing or tcp_open(str(host), int(port)):
                deadline = time.time() + 8.0
                while time.time() < deadline:
                    if is_rag_up(host, port, timeout=1.0):
                        self._log(f"RAG API ready at {url}")
                        self.panel.set_rag_running(True, f"{url}/v1")
                        return
                    time.sleep(0.4)
                if existing:
                    self._log(f"RAG API starting (PIDs {', '.join(str(p) for p in existing)})")
                    self.panel.set_rag_running(True, f"{url}/v1")
                    return
            if not locked:
                self._log("Another window is starting the RAG API — waiting…")
                QMessageBox.information(
                    self.parent,
                    "RAG API",
                    "Another process is starting the RAG API. Retry in a few seconds.",
                )
                return

            creationflags = 0
            if sys.platform == "win32":
                creationflags = subprocess.CREATE_NEW_CONSOLE  # type: ignore[attr-defined]
            env = os.environ.copy()
            env["PYTHONPATH"] = str(ROOT) + os.pathsep + env.get("PYTHONPATH", "")
            try:
                proc = subprocess.Popen(
                    [str(py), "-u", str(script)],
                    cwd=str(ROOT),
                    env=env,
                    creationflags=creationflags,
                )
                _SPAWNED.append(proc)
            except OSError as e:
                self._log(f"Failed to launch RAG API: {e}")
                QMessageBox.critical(self.parent, "RAG API", f"Failed to launch:\n{e}")
                return

            deadline = time.time() + 12.0
            while time.time() < deadline:
                if is_rag_up(host, port, timeout=1.0):
                    self._log(f"RAG API running at {url}/v1 · Web UI {url}/")
                    self.panel.set_rag_running(True, f"{url}/v1")
                    return
                if proc.poll() is not None:
                    break
                time.sleep(0.4)

        if is_rag_up(host, port):
            self.panel.set_rag_running(True, f"{url}/v1")
            return

        self._log(f"Started _start_rag_api.py but {url}/health is not answering yet")
        QMessageBox.warning(
            self.parent,
            "RAG API",
            f"RAG API process started but {url}/health is not answering yet.\n\n"
            "Check the RAG console window for errors (embeddings :8080, chat :8081).",
        )

    def stop_rag(self) -> None:
        host, port = rag_bind(self.config)
        stopped = 0
        for proc in list(_SPAWNED):
            if proc.poll() is not None:
                continue
            try:
                proc.terminate()
                try:
                    proc.wait(timeout=3)
                except subprocess.TimeoutExpired:
                    proc.kill()
                stopped += 1
            except Exception:
                pass
        _SPAWNED.clear()
        stopped += len(kill_script_pids(RAG_SCRIPT))

        self.config.setdefault("rag_api", {})["enabled"] = False
        self._save_config()
        if self.panel is not None:
            self.panel.set_rag_running(False)
        if stopped:
            self._log(f"Stopped RAG API ({stopped} process(es))")
        else:
            self._log("RAG API was not running")

        if is_rag_up(host, port):
            self._log(f"Note: something still responds on {rag_url(host, port)}/health")

    def clear_embeddings(db_path: str = "") -> Dict[str, Any]:
    """Null document vectors and clear kb_embedding_corpus. Keep document rows."""
    from WarpCore.ship_schema import ensure_ship_computer_schema
    from WarpCore.search import build_documents_vec

    path = _db_path(db_path)
    ensure_ship_computer_schema(path)
    from spock_rag_adapter.embedding_corpus import ensure_schema

    ensure_schema(path)

    conn = sqlite3.connect(path, timeout=60)
    try:
        before_emb = int(
            conn.execute(
                "SELECT COUNT(*) FROM documents WHERE embedding IS NOT NULL"
            ).fetchone()[0]
        )
        before_docs = int(conn.execute("SELECT COUNT(*) FROM documents").fetchone()[0])
        try:
            before_corpus = int(
                conn.execute("SELECT COUNT(*) FROM kb_embedding_corpus").fetchone()[0]
            )
        except sqlite3.OperationalError:
            before_corpus = 0

        conn.execute(
            """
            UPDATE documents
            SET embedding = NULL,
                embedding_model_path = NULL,
                embedding_dim = NULL
            WHERE embedding IS NOT NULL
               OR embedding_model_path IS NOT NULL
               OR embedding_dim IS NOT NULL
            """
        )
        cleared_docs = conn.total_changes
        try:
            conn.execute("DELETE FROM kb_embedding_corpus")
        except sqlite3.OperationalError:
            pass
        conn.commit()
        build_documents_vec(path)

        after_emb = int(
            conn.execute(
                "SELECT COUNT(*) FROM documents WHERE embedding IS NOT NULL"
            ).fetchone()[0]
        )
        try:
            after_corpus = int(
                conn.execute("SELECT COUNT(*) FROM kb_embedding_corpus").fetchone()[0]
            )
        except sqlite3.OperationalError:
            after_corpus = 0
    finally:
        conn.close()

    return {
        "ok": True,
        "db_path": path,
        "documents_total": before_docs,
        "embeddings_before": before_emb,
        "embeddings_after": after_emb,
        "rows_touched": cleared_docs,
        "corpus_before": before_corpus,
        "corpus_after": after_corpus,
    }

    def embedding_status(
    db_path: str = "",
    embeddings_url: str = "http://localhost:8080/embeddings",
) -> Dict[str, Any]:
    """Counts + corpus row + live :8080 probe."""
    from WarpCore.ship_schema import ensure_ship_computer_schema
    from spock_rag_adapter.embedding_corpus import (
        ensure_schema,
        probe_live_model,
        read_corpus,
    )

    path = _db_path(db_path)
    out: Dict[str, Any] = {
        "db_path": path,
        "db_exists": os.path.isfile(path),
        "documents": 0,
        "with_embedding": 0,
        "without_embedding": 0,
        "corpus": None,
        "live": None,
    }
    if not out["db_exists"]:
        out["live"] = probe_live_model(embeddings_url)
        return out

    ensure_ship_computer_schema(path)
    ensure_schema(path)
    conn = sqlite3.connect(path, timeout=30)
    try:
        out["documents"] = int(
            conn.execute("SELECT COUNT(*) FROM documents").fetchone()[0]
        )
        out["with_embedding"] = int(
            conn.execute(
                "SELECT COUNT(*) FROM documents WHERE embedding IS NOT NULL"
            ).fetchone()[0]
        )
        out["without_embedding"] = out["documents"] - out["with_embedding"]
    except sqlite3.OperationalError as e:
        out["error"] = str(e)
    finally:
        conn.close()

    out["corpus"] = read_corpus(path)
    out["live"] = probe_live_model(embeddings_url)
    return out
