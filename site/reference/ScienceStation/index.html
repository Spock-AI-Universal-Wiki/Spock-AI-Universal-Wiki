    class ScienceStationWindow(SuiteShellWindow):
    app_title = "Spock-Science-Station"
    app_subtitle = (
        "Hardware → HF Models → Launch embed/chat → RUN Wiki Agent → "
        "Ship's Logs → Ask AI."
    )

    def __init__(self, parent=None):
        ensure_root_on_path()
        self.state = ModelSelectorState()
        self.server_manager = LlamaServerManager()
        self.manager = self.server_manager  # Ship Status / Launch compatibility
        self._steps: Dict[str, Any] = {}
        self._hw_scanned_once = False
        self._logs = None
        self._ask = None
        super().__init__(parent)

    def on_ready(self) -> None:
        self._build_steps()
        self.show_home()

    def build_home_widget(self) -> QWidget:
        return build_markdown_home(
            self.theme,
            HELPER_MD,
            fallback_title=self.app_title,
            fallback_subtitle=self.app_subtitle,
        )

    def show_home(self) -> None:
        reload_markdown_home(self._home, self.theme)
        super().show_home()

    def nav_steps(self) -> Sequence[NavItem]:
        # AI Helper / Ship Status / About / Google Search live on the right Tools dock.
        return [
            ("Home", self.show_home),
            ("1 Hardware", self._show_hardware),
            ("2 HF Models", self._show_models),
            ("3 Launch", self._show_launch),
            ("4 Run Wiki Scan", lambda: self.show_center(self._steps["run"])),
            ("5 Ship's Logs", self._show_ships_logs),
            ("6 Run Ask AI", self._show_ask_ai),
            ("-----", None),
            ("Exit", self.close),
        ]

    def _show_ships_logs(self) -> None:
        """Captains Log / Agents Queue / Documents / Quarantine, tabbed."""
        if self._logs is None:
            self._logs = build_ships_logs_panel(self.theme, SHIP_COMPUTER_DB)
            self.stack.addWidget(self._logs)
        select = getattr(self._logs, "select_table", None)
        if callable(select):
            select("documents")
        self.show_center(self._logs)

    def _show_ask_ai(self) -> None:
        if self._ask is None:
            from EngineeringDeck.ask_panel import AskAiPanel
            from WarpCore.search import hybrid_search

            self._ask = AskAiPanel(
                self.theme,
                get_llm=lambda: make_llm_client(self.config),
                hybrid_search_fn=hybrid_search,
                get_emb_url=lambda: merge_embedding_profile(
                    self.config.get("embedding_profile")
                ).get("embeddings_url", "http://localhost:8080/embeddings"),
            )
            self.stack.addWidget(self._ask)
        self.show_center(self._ask)
        self._maybe_offer_ships_path_to_ask()

    def _maybe_offer_ships_path_to_ask(self) -> None:
        if self._logs is None or self._ask is None:
            return
        if not hasattr(self, "_offered_ask_path_filters"):
            self._offered_ask_path_filters = set()
        from spock_shell.ask_path_handoff import offer_copy_ships_path_to_ask

        offer_copy_ships_path_to_ask(
            self,
            logs=self._logs,
            ask=self._ask,
            db_path=SHIP_COMPUTER_DB,
            offered=self._offered_ask_path_filters,
        )

    def build_right_dock(self, layout: QVBoxLayout) -> None:
        search = SidePanel("Search", self.theme)
        search.add_button("Google Search", self._show_brave_search)
        layout.addWidget(search)
        super().build_right_dock(layout)

    def _openai_compat_from_config(self) -> Dict[str, Any]:
        """Slice Ask AI OpenAI-compatible fields for the Gemini (Vertex) tab."""
        c = self.config
        return {
            "openai_compat_base_url": (c.get("openai_compat_base_url") or "").strip(),
            "openai_compat_api_key": c.get("openai_compat_api_key") or "",
            "openai_compat_model": (c.get("openai_compat_model") or "").strip(),
            "openai_compat_n_ctx": c.get("openai_compat_n_ctx"),
            "openai_compat_max_tokens": c.get("openai_compat_max_tokens"),
        }

    def _on_gemini_use_for_ask_ai(self, model_id: str) -> None:
        mid = (model_id or "").strip()
        if not mid:
            return
        self.config["openai_compat_model"] = mid
        self.config["ask_ai_provider"] = "OpenAI-compatible"
        save_config(self.config)

    def _build_steps(self) -> None:
        self._steps["hardware"] = build_hardware_step(self.theme, self.state)
        models = ModelsPanel(
            self.theme,
            self.state,
            openai_compat=self._openai_compat_from_config(),
        )
        models.request_show_info.connect(self._show_model_info_for)
        models.gemini_use_for_ask_ai.connect(self._on_gemini_use_for_ask_ai)
        self._steps["models"] = models
        launch = LaunchPanel(self.theme, self.state, self.server_manager)
        launch.request_model_info.connect(self._show_model_info)
        self._steps["launch"] = launch
        self._steps["model_info"] = ModelInfoPanel(self.theme, self.state)
        self._steps["run"] = build_wiki_agent_run_panel(
            self.theme, config=self.config
        )
        self._steps["brave"] = build_brave_search_panel(self.theme)

        hw_panel = getattr(self._steps["hardware"], "_panel", None)
        if hw_panel is not None:
            hw_panel.scan_complete.connect(
                lambda _specs: self._on_hardware_scanned()
            )
            # Keep Launch GPU list in sync after scan (same as Navigation).
            hw_panel.scan_complete.connect(lambda _specs: launch._populate_gpus())

        for w in self._steps.values():
            self.stack.addWidget(w)

        # If state already has specs (restored), unlock HF Query buttons.
        if self.state.hardware_specs:
            self._enable_hf_query_buttons()

    def _on_hardware_scanned(self) -> None:
        self._enable_hf_query_buttons()

    def _enable_hf_query_buttons(self) -> None:
        models = self._steps.get("models")
        if models is None:
            return
        enabled = bool(self.state.hardware_specs)
        models.query_btn.setEnabled(enabled)
        models.query_embed_btn.setEnabled(enabled)

    def _show_hardware(self) -> None:
        w = self._steps["hardware"]
        self.show_center(w)
        panel = getattr(w, "_panel", None)
        if panel is not None and not self._hw_scanned_once:
            self._hw_scanned_once = True
            QTimer.singleShot(0, panel._start_scan)

    def _show_models(self) -> None:
        self._enable_hf_query_buttons()
        models = self._steps.get("models")
        if models is not None:
            models.set_openai_compat(self._openai_compat_from_config())
        self.show_center(self._steps["models"])

    def _show_launch(self) -> None:
        launch = self._steps["launch"]
        launch._populate_gpus()
        launch._update_buttons()
        self.show_center(launch)

    def _show_model_info(self) -> None:
        self._show_model_info_for(self.state.selected_hf_model_id)

    def _show_model_info_for(self, model_id: str = "") -> None:
        mid = (model_id or self.state.selected_hf_model_id or "").strip()
        panel = self._steps["model_info"]
        if mid:
            panel.load_model(mid)
        else:
            panel._refresh_from_state()
        self.show_center(panel)

    def _show_brave_search(self) -> None:
        self.show_center(self._steps["brave"])

    def closeEvent(self, event) -> None:  # noqa: N802
        run = self._steps.get("run")
        stop = getattr(run, "_stop", None) if run is not None else None
        if callable(stop):
            try:
                stop()
            except Exception:
                pass
        super().closeEvent(event)

    class ShipsLogsPanel(QWidget):
    """Tabbed browser over captains_log, agent_queue, documents, quarantine."""

    def __init__(self, theme: Dict[str, str], db_path: str, parent=None):
        super().__init__(parent)
        self.theme = theme
        self.db_path = db_path
        self._db = None
        self._loaded: Dict[int, QWidget] = {}

        lay = QVBoxLayout(self)
        tip = QLabel(
            "<b>Ship's Logs</b>: browse <code>ship_computer.db</code>. Search or "
            "filter a table, then Preview (or double-click a path cell) to open "
            "the source file."
        )
        tip.setWordWrap(True)
        lay.addWidget(tip)

        self.tabs = QTabWidget()
        for label, _table in LOG_TABLES:
            self.tabs.addTab(_message(f"Loading {label}…"), label)
        self.tabs.currentChanged.connect(self._ensure_tab)
        lay.addWidget(self.tabs, 1)

        # Open on Documents (can be large; lazy-load still applies).
        docs_index = next(
            (i for i, (_label, table) in enumerate(LOG_TABLES) if table == "documents"),
            0,
        )
        self.tabs.setCurrentIndex(docs_index)
        self._ensure_tab(docs_index)

    def _ensure_db(self) -> bool:
        if self._db is not None:
            return True
        from db_browser import DatabaseManager

        try:
            self._db = DatabaseManager(self.db_path)
        except Exception:
            self._db = None
            return False
        return True

    def _ensure_tab(self, index: int) -> None:
        if index < 0 or index >= len(LOG_TABLES) or index in self._loaded:
            return
        label, table = LOG_TABLES[index]
        widget = self._build_browser(label, table)
        self._loaded[index] = widget
        self.tabs.removeTab(index)
        self.tabs.insertTab(index, widget, label)
        self.tabs.setCurrentIndex(index)

    def _build_browser(self, label: str, table: str) -> QWidget:
        if not self._ensure_db():
            return _message(f"Could not open the database:\n{self.db_path}")
        from db_browser import TableBrowser

        try:
            return TableBrowser(self._db, table, self.theme)
        except Exception as e:
            return _message(f"Could not open table <b>{table}</b>:<br>{e}")

    def browser(self, table: str) -> Optional[QWidget]:
        """Loaded browser for a table name, or None if that tab is untouched."""
        for index, (_label, name) in enumerate(LOG_TABLES):
            if name == table:
                return self._loaded.get(index)
        return None

    def documents_path_filter(self) -> str:
        """Path: field on the Documents table browser (empty if not loaded)."""
        browser = self.browser("documents")
        if browser is None:
            return ""
        filt = getattr(browser, "filter_input", None)
        if filt is None:
            return ""
        return str(filt.text() or "").strip()

    def select_table(self, table: str) -> None:
        """Switch to a log table tab (lazy-builds the browser)."""
        for index, (_label, name) in enumerate(LOG_TABLES):
            if name == table:
                self.tabs.setCurrentIndex(index)
                self._ensure_tab(index)
                return

    class ProcessOutputReader(QThread):
    """Reads subprocess stdout line-by-line on a worker thread."""

    line_read = pyqtSignal(str)
    finished_rc = pyqtSignal(int)

    def __init__(self, process: subprocess.Popen, parent=None):
        super().__init__(parent)
        self.process = process
        self._running = True

    def run(self) -> None:
        assert self.process.stdout is not None
        stream = io.TextIOWrapper(
            self.process.stdout, encoding="utf-8", errors="replace"
        )
        try:
            while self._running:
                line = stream.readline()
                if not line:
                    break
                self.line_read.emit(strip_ansi(line.rstrip("\n")))
        except Exception as e:
            self.line_read.emit(f"[reader] {e}")
        finally:
            try:
                rc = self.process.wait(timeout=2)
            except Exception:
                rc = self.process.poll()
                if rc is None:
                    rc = -1
            self.finished_rc.emit(int(rc if rc is not None else -1))

    def stop(self) -> None:
        self._running = False

    theme: Dict[str, str],

    """Center pane: Run Wiki Scan / Stop Scan + Embedding Status + live log.

    ``title``/``source_label`` let other mini-apps host this step under their

    """
    wrap = QWidget()
    lay = QVBoxLayout(wrap)

    lay.addWidget(title_lbl)

    # Controls at top of middle pane
    row = QHBoxLayout()

    state: Dict[str, Any] = {"process": None, "reader": None, "once": True}

    def _append(line: str) -> None:
        clean = strip_ansi(line)
        if not clean and line:
            return
        log.append(clean)
        sb = log.verticalScrollBar()
        sb.setValue(sb.maximum())
        if "SCAN_COMPLETE" in clean:
            status.setText("Scan complete — finishing…")

    def _on_done(rc: int) -> None:
        if state.get("once") and rc == 0:
            status.setText("Scan complete")
        elif state.get("once") and rc != 0:
            status.setText(f"Scan ended (exit {rc})")
        else:
            status.setText(f"Stopped (exit {rc})")
        run_btn.setEnabled(True)
        stop_btn.setEnabled(False)
        continuous.setEnabled(True)
        state["process"] = None
        state["reader"] = None
        set_busy(progress, False)

    def _stop() -> None:
        reader = state.get("reader")
        proc = state.get("process")
        if reader is not None:
            reader.stop()
        pid = None
        if proc is not None and proc.poll() is None:
            try:
                pid = int(proc.pid)
            except (TypeError, ValueError, AttributeError):
                pid = None
        from spock_shell.process_tree import stop_wiki_scan

        info = stop_wiki_scan(
            ROOT,
            proc_pid=pid,
            db_path=SHIP_COMPUTER_DB,
        )
        n = len(info.get("killed_pids") or [])
        q = int(info.get("queue_cancelled") or 0)
        _append(
            f"[{source_label}] Wiki scan stopped "
            f"(killed {n} process tree(s), cancelled {q} queue job(s))."
        )
        state["process"] = None
        state["reader"] = None
        run_btn.setEnabled(True)
        stop_btn.setEnabled(False)
        continuous.setEnabled(True)
        status.setText("Stopped")
        set_busy(progress, False)
        from spock_shell.busy_ui import clear_override_cursors

        clear_override_cursors()

    def _run() -> None:
        if state.get("process") is not None and state["process"].poll() is None:
            QMessageBox.information(
                wrap, "Wiki Scan", "Already running — use Stop Scan first."
            )
            return
        script = ROOT / "wiki_agent.py"
        py = ROOT / ".venv" / "Scripts" / "python.exe"
        exe = str(py) if py.is_file() else sys.executable
        if not script.is_file():
            QMessageBox.warning(wrap, "Wiki Scan", f"Missing {script}")
            return

        env = _embedding_env(config)
        emb = env.get("EMBEDDINGS_URL", "")
        once = not continuous.isChecked()
        state["once"] = once
        mode_flag = "--once" if once else "--continuous"
        try:
            from spock_shell.embedding_corpus import record_from_live_server

            profile = {}
            try:
                from model_selector_core import merge_embedding_profile

                profile = merge_embedding_profile((config or {}).get("embedding_profile"))
            except Exception:
                profile = {}
            ok, msg, row = record_from_live_server(
                emb or "http://localhost:8080/embeddings",
                embeddings_dim=int(profile.get("embeddings_dim") or env.get("EMBEDDINGS_DIM") or 4096),
                model_id=str(profile.get("model_id") or ""),
            )
            if not ok:
                QMessageBox.warning(wrap, "Wiki Scan", msg)
                return
            live_path = str((row or {}).get("model_path") or "")
            if live_path:
                env["SPOCK_EMBED_MODEL_PATH"] = live_path
            log.clear()
            _append(f"$ {exe} -u {script} {mode_flag}")
            _append(f"EMBEDDINGS_URL={emb}")
            _append(f"Embedding GGUF: {(row or {}).get('model_name') or live_path or '—'}")
            _append(msg)
            if once:
                _append(
                    "(scan-once — exits when queue is idle after bootstrap + update_index)\n"
                )
            else:
                _append(
                    "(continuous watch — STATUS while idle; use Stop Scan to exit)\n"
                )
        except Exception as e:
            QMessageBox.warning(
                wrap,
                "Wiki Scan",
                f"Could not record the live embedding model:\n{e}",
            )
            return

        try:
            proc = subprocess.Popen(
                [exe, "-u", str(script), mode_flag],
                cwd=str(ROOT),
                stdout=subprocess.PIPE,
                stderr=subprocess.STDOUT,
                env=env,
                bufsize=0,
            )
        except Exception as e:
            QMessageBox.critical(wrap, "Wiki Scan", str(e))
            return

        state["process"] = proc
        reader = ProcessOutputReader(proc, wrap)
        reader.line_read.connect(_append)
        reader.finished_rc.connect(_on_done)
        state["reader"] = reader
        reader.start()
        mode_lbl = "scan-once" if once else "continuous"
        status.setText(f"Running ({mode_lbl}, PID {proc.pid}) · {emb}")
        run_btn.setEnabled(False)
        stop_btn.setEnabled(True)
        continuous.setEnabled(False)
        set_busy(progress, True)

    def _embedding_status() -> None:
        script = ROOT / "wiki_agent_status.py"
        if not script.is_file():
            QMessageBox.warning(wrap, "Embedding Status", f"Not found: {script}")
            return
        py = ROOT / ".venv" / "Scripts" / "python.exe"
        exe = str(py) if py.is_file() else sys.executable
        QApplication.setOverrideCursor(Qt.CursorShape.WaitCursor)
        try:
            result = subprocess.run(
                [exe, str(script)],
                capture_output=True,
                text=True,
                encoding="utf-8",
                errors="replace",
                timeout=120,
                cwd=str(ROOT),
                env=_embedding_env(config),
            )
            output = (result.stdout or "") + (
                f"\n{result.stderr}" if result.stderr else ""
            )
            _show_output_dialog(
                wrap,
                "Embedding Status (wiki_agent_status.py)",
                strip_ansi(output.strip() or "(no output)"),
            )
        except subprocess.TimeoutExpired:
            QMessageBox.warning(
                wrap, "Timeout", "wiki_agent_status.py timed out after 120s"
            )
        except Exception as e:
            QMessageBox.warning(wrap, "Embedding Status", str(e))
        finally:
            QApplication.restoreOverrideCursor()

    def _clear_embeddings() -> None:
        proc = state.get("process")
        if proc is not None and proc.poll() is None:
            QMessageBox.information(
                wrap,
                "Clear Embeddings",
                "Stop the running Wiki Scan first, then clear embeddings.",
            )
            return
        reply = QMessageBox.question(
            wrap,
            "Clear Embeddings (re-embed)",
            "Clear all document embeddings and kb_embedding_corpus?\n\n"
            "This keeps documents, settings, and profiles.\n"
            "It does NOT wipe captains_log / queue (unlike Reset content).\n\n"
            "Afterward: launch the new embedding GGUF on :8080, then Run Wiki Scan.",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
            QMessageBox.StandardButton.No,
        )
        if reply != QMessageBox.StandardButton.Yes:
            status.setText("Clear embeddings cancelled")
            return
        from spock_shell.busy_ui import busy_scope, clear_override_cursors

        try:
            from spock_rag_adapter.reembed import clear_embeddings, format_clear_result

            with busy_scope(wrap, "Clearing embeddings…"):
                result = clear_embeddings(SHIP_COMPUTER_DB)
            clear_override_cursors()
            _show_output_dialog(
                wrap, "Clear Embeddings (re-embed)", format_clear_result(result)
            )
            status.setText(
                f"Cleared embeddings "
                f"({result.get('embeddings_before')} → {result.get('embeddings_after')})"
            )
            _append(
                f"[Clear Embeddings] {result.get('embeddings_before')} → "
                f"{result.get('embeddings_after')} "
                f"(docs kept: {result.get('documents_total')})"
            )
        except Exception as e:
            clear_override_cursors()
            QMessageBox.warning(wrap, "Clear Embeddings", str(e))
            status.setText(f"Clear embeddings error: {e}")
